cmake_minimum_required(VERSION 3.21)
project(biolab LANGUAGES C ASM)
file(GLOB _sources src/*.c src/gen/*.c)
psn00bsdk_add_executable(biolab GPREL ${_sources})
target_include_directories(biolab PRIVATE src src/gen)
target_compile_options(biolab PRIVATE -O2 -Wall -Wno-unused-function -Wno-unused-variable)
if(DEFINED DEBUG_SEED)
	target_compile_definitions(biolab PRIVATE DEBUG_SEED=${DEBUG_SEED} DEBUG_RCOUNT=${DEBUG_RCOUNT})
endif()
if(DEFINED DEBUG_LEVEL)
	target_compile_definitions(biolab PRIVATE DEBUG_LEVEL=${DEBUG_LEVEL} DEBUG_X=${DEBUG_X} DEBUG_Y=${DEBUG_Y})
endif()
